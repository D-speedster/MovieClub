مشکل: کد فعلی از movie.genre (رشته) می‌خواند اما مدل جدید genres (آرایه) دارد.

کد فعلی (اشتباه):
  filtered = filtered.filter(movie => movie.genre && movie.genre.includes(filters.genre));

کد صحیح:
  filtered = filtered.filter(movie => {
    const genres = movie.genres || movie.genre || [];
    return Array.isArray(genres)
      ? genres.includes(filters.genre)
      : genres.split(',').map(g => g.trim()).includes(filters.genre);
  });

همچنین در همین فایل، مرتب‌سازی از CreatedAt (با C بزرگ) استفاده می‌کند:

--- B2: Controller نظرات ---

فایل جدید: Backend/controller/comment.js

const Comment = require('../models/comment');
const ContentSchema = require('../models/content');

// GET /comments/:contentId — دریافت نظرات یک فیلم
exports.GetComments = async (req, res, next) => {
    try {
        const comments = await Comment.find({
            contentId: req.params.contentId,
            status: 'approved'
        }).sort({ createdAt: -1 }).limit(50);
        res.json(comments);
    } catch (err) { next(err); }
};

// POST /comments — ارسال نظر (نیاز به login)
exports.PostComment = async (req, res, next) => {
    try {
        const { contentId, text, rating } = req.body;
        if (!contentId || !text) {
            return res.status(400).json({ message: 'contentId و text الزامی هستند' });
        }
        const content = await ContentSchema.findById(contentId);
        if (!content) return res.status(404).json({ message: 'محتوا یافت نشد' });

        const comment = await Comment.create({
            contentId,
            userId: req.user.userId,
            username: req.user.name,
            text,
            rating: rating || null
        });
        res.status(201).json({ message: 'نظر ثبت شد', data: comment });
    } catch (err) { next(err); }
};

// GET /comments/admin/all — همه نظرات برای ادمین
exports.GetAllComments = async (req, res, next) => {
    try {
        const comments = await Comment.find({})
            .populate('contentId', 'title')
            .sort({ createdAt: -1 });
        res.json(comments);
    } catch (err) { next(err); }
};

// PUT /comments/:id/status — تغییر وضعیت نظر (ادمین)
exports.UpdateCommentStatus = async (req, res, next) => {
    try {
        const { status } = req.body;
        const updated = await Comment.findByIdAndUpdate(
            req.params.id, { status }, { new: true }
        );
        if (!updated) return res.status(404).json({ message: 'نظر یافت نشد' });
        res.json({ message: 'وضعیت نظر تغییر کرد', data: updated });
    } catch (err) { next(err); }
};

// DELETE /comments/:id — حذف نظر (ادمین)
exports.DeleteComment = async (req, res, next) => {
    try {
        const deleted = await Comment.findByIdAndDelete(req.params.id);
        if (!deleted) return res.status(404).json({ message: 'نظر یافت نشد' });
        res.json({ message: 'نظر حذف شد' });
    } catch (err) { next(err); }
};

  const ProtectedRoute = ({ children, adminOnly = false }) => {
      const token = localStorage.getItem('token');
      const role = localStorage.getItem('role');

      if (!token) return <Navigate to="/auth/login" replace />;
      if (adminOnly && role !== 'Admin' && role !== 'Owner') {
          return <Navigate to="/" replace />;
      }
      return children;
  };

--- F2: Pagination در API ---

فایل: Backend/controller/content.js
برای GetContent و GetSeriesList، pagination اضافه کن:

  exports.GetContent = async (req, res, next) => {
      try {
          const page = parseInt(req.query.page) || 1;
          const limit = parseInt(req.query.limit) || 20;
          const skip = (page - 1) * limit;

          const [movies, total] = await Promise.all([
              ContentSchema.find({ type: 'movie' }).skip(skip).limit(limit),
              ContentSchema.countDocuments({ type: 'movie' })
          ]);

          res.json({
              data: movies,
              pagination: {
                  page,
                  limit,
                  total,
                  totalPages: Math.ceil(total / limit)
              }
          });
      } catch (err) { next(err); }
  };
